"""Fixed local subset of WrongSecrets; no caller-selected destinations or tunnels."""
import http.client
import http.server
import os
import re
import threading
import time
from urllib.parse import urlsplit

target = os.environ['ASCEND_TARGET']
if not re.fullmatch(r'ascend-ll-[a-f0-9]{8}-[a-f0-9]{16}-app', target):
    raise ValueError('Unexpected lab target')
home = b'''<!doctype html><html lang="en"><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><title>WrongSecrets local practice</title><style>body{font:1.1rem system-ui;max-width:48rem;margin:3rem auto;padding:0 1rem;line-height:1.6}li{margin:1rem 0}</style><h1>WrongSecrets local practice</h1><p>Find three ways an application exposes a secret. Open a challenge, examine the downloaded source files, and record your answers and explanation in the work directory shown by Ascend.</p><ol><li><a href="/challenge/challenge-1">Hardcoded source secret</a></li><li><a href="/challenge/challenge-2">Application configuration</a></li><li><a href="/challenge/challenge-3">Container environment</a></li></ol><p>This recipe includes these three challenges. Other challenges in the upstream navigation need a separate environment.</p><p>Save answers in answers.json and observations in notes.txt. Ascend retains these files across stop/resume; the application's browser badges reset when it restarts. Run local-lab check to replay your saved answers.</p><p><a href="https://github.com/OWASP/wrongsecrets/tree/7144c57fb15708f9486dab83d536448302d4e51b" rel="noreferrer">WrongSecrets v1.14.2 source and AGPL-3.0 license</a></p></html>'''

class Handler(http.server.BaseHTTPRequestHandler):
    def setup(self):
        super().setup()
        self.connection.settimeout(5)

    def log_message(self, *args):
        pass

    def reply(self, status, data, content_type='text/plain; charset=utf-8', headers=()):
        self.send_response(status)
        self.send_header('Content-Type', content_type)
        self.send_header('Content-Length', str(len(data)))
        self.send_header('Connection', 'close')
        self.send_header('X-Content-Type-Options', 'nosniff')
        self.send_header('Cache-Control', 'no-store')
        self.send_header('Content-Security-Policy', "frame-ancestors 'none'; form-action 'self'; connect-src 'self'")
        for key, value in headers:
            self.send_header(key, value)
        self.end_headers()
        if self.command != 'HEAD':
            self.wfile.write(data)
        self.close_connection = True

    def forward(self):
        host = self.headers.get('Host', '')
        if not re.fullmatch(r'127\.0\.0\.1:[0-9]{1,5}', host):
            return self.reply(403, b'Use the loopback URL shown by Ascend.')
        origin = self.headers.get('Origin')
        referer = self.headers.get('Referer')
        if (self.headers.get('Sec-Fetch-Site') == 'cross-site' or
            (origin and (urlsplit(origin).scheme != 'http' or urlsplit(origin).netloc != host)) or
            (referer and urlsplit(referer).netloc != host)):
            return self.reply(403, b'Cross-origin lab access is not allowed.')
        parts = urlsplit(self.path)
        path = parts.path
        if parts.scheme or parts.netloc or '%' in path or '\\' in path or '..' in path.split('/') or self.path.startswith('//'):
            return self.reply(400, b'Unsupported local path.')
        if self.headers.get('Transfer-Encoding') or len(self.headers.get_all('Content-Length', [])) > 1:
            return self.reply(400, b'Unsupported request framing.')
        try:
            size = int(self.headers.get('Content-Length', '0'))
        except ValueError:
            return self.reply(400, b'Invalid content length.')
        if not 0 <= size <= 65536:
            return self.reply(413, b'Lab requests must fit within 64 KiB.')
        challenge = bool(re.fullmatch(r'/challenge/challenge-[123]', path))
        reading = bool(re.fullmatch(r'/(?:spoil/challenge-[123]|actuator/health|favicon.ico)', path))
        static = bool(re.fullmatch(r'/(?:webjars|css|js|images|img|fonts)/[a-zA-Z0-9_./@-]+', path))
        if path == '/' and self.command in {'GET', 'HEAD'}:
            return self.reply(200, home, 'text/html; charset=utf-8')
        if not challenge and not (self.command in {'GET', 'HEAD'} and (reading or static)):
            return self.reply(404, b'This recipe includes challenges 1, 2 and 3. Return to the lab home page.')
        connection = http.client.HTTPConnection(target, 8080, timeout=5)
        try:
            body = self.rfile.read(size) if size else None
            if body is not None and len(body) != size:
                return self.reply(400, b'Incomplete request body.')
            headers = {key: self.headers[key] for key in ('Content-Type', 'Cookie', 'Accept') if self.headers.get(key)}
            headers['Host'] = host
            headers['User-Agent'] = 'Ascend-local-lab'
            connection.request(self.command, self.path, body, headers)
            response = connection.getresponse()
            limit = 4194304 if static else 262144
            data = response.read(limit + 1)
            if len(data) > limit:
                return self.reply(502, b'Lab response exceeds its allowed size.')
            returned = [(key, value) for key, value in response.getheaders() if key.lower() == 'set-cookie']
            location = response.getheader('Location')
            if location and location.startswith('/') and not location.startswith('//'):
                returned.append(('Location', location))
            self.reply(response.status, data, response.getheader('Content-Type', 'application/octet-stream'), returned)
        except (OSError, http.client.HTTPException, ValueError):
            self.reply(502, b'The local lab is not ready. Check its status with Ascend.')
        finally:
            connection.close()

    do_GET = do_HEAD = do_POST = forward

server = http.server.HTTPServer(('0.0.0.0', 8080), Handler)
threading.Thread(target=server.serve_forever, daemon=True).start()
while time.time() < float(os.environ['ASCEND_LAB_EXPIRES']):
    time.sleep(.5)
os._exit(0)
